// Seeded random numbers, so "New summer" / "New region" are repeatable and
// the picture doesn't change when only the sampling design changes.
// (mulberry32: a tiny, well-known 32-bit generator.)
rng = seed => {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// A standard normal draw (Box-Muller)
rnorm = r => {
  let u = 0;
  while (u === 0) u = r();
  return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * r());
}

// The "textbook" 95% CI for a mean: mean ± t * s / sqrt(n).
// This is exactly what lm(y ~ 1) and t.test(y) report. It assumes every
// observation is independent, which is the assumption this page breaks.
naiveCI = (values, tcrit) => {
  const n = values.length;
  const m = d3.mean(values);
  const se = d3.deviation(values) / Math.sqrt(n);
  return {m, se, lo: m - tcrit * se, hi: m + tcrit * se};
}

// Summarize many repeats of the same design.
//   reps: [{err, half, popVar}] where err = estimate - truth,
//   half = CI half-width, popVar = variance over the whole target period/area.
summarize = (reps) => {
  const covered = reps.filter(d => Math.abs(d.err) <= d.half).length / reps.length;
  // Half-width that would really have caught 95% of the truths
  const actual95 = d3.quantile(reps.map(d => Math.abs(d.err)), 0.95);
  const claimed = d3.mean(reps, d => d.half);
  // Effective sample size: how many *randomly placed* readings would give
  // the same typical error (variance of one random reading / mean squared error)
  const mse = d3.mean(reps, d => d.err * d.err);
  const nEff = d3.mean(reps, d => d.popVar) / mse;
  return {covered, actual95, claimed, nEff};
}

// Coverage plot shared by both widgets: one vertical CI per repeat, drawn as
// error (estimate - truth). A bar that crosses the zero line caught the truth.
coveragePlot = (reps, {w, unit, ylim = 4, nShow = 100}) => {
  const h = 210, m = {t: 14, r: 12, b: 30, l: 46};
  const shown = reps.slice(0, nShow);
  const x = d3.scaleLinear().domain([0, shown.length]).range([m.l, w - m.r]);
  const y = d3.scaleLinear().domain([-ylim, ylim]).range([h - m.b, m.t]).clamp(true);
  const svg = d3.create("svg").attr("viewBox", [0, 0, w, h]).style("width", "100%").style("max-width", w + "px");
  svg.append("g").attr("transform", `translate(${m.l},0)`)
    .call(d3.axisLeft(y).ticks(5).tickFormat(d => (d > 0 ? "+" : "") + d))
    .call(g => g.selectAll("text").attr("font-size", 11));
  svg.append("text").attr("transform", `translate(12,${(h - m.b + m.t) / 2}) rotate(-90)`)
    .attr("text-anchor", "middle").attr("font-size", 11).attr("fill", col.soft)
    .text(`estimate − truth (${unit})`);
  svg.append("text").attr("x", (m.l + w - m.r) / 2).attr("y", h - 6).attr("text-anchor", "middle")
    .attr("font-size", 11).attr("fill", col.soft).text(`${shown.length} repeats of the same design`);
  // Zero line = the truth
  svg.append("line").attr("x1", m.l).attr("x2", w - m.r).attr("y1", y(0)).attr("y2", y(0))
    .attr("stroke", col.ink).attr("stroke-dasharray", "4 3");
  const bw = Math.max(1.2, (x(1) - x(0)) * 0.55);
  shown.forEach((d, i) => {
    const hit = Math.abs(d.err) <= d.half;
    const cx = x(i + 0.5);
    svg.append("line").attr("x1", cx).attr("x2", cx)
      .attr("y1", y(d.err - d.half)).attr("y2", y(d.err + d.half))
      .attr("stroke", hit ? col.blue : col.red).attr("stroke-width", bw).attr("stroke-linecap", "round");
    svg.append("circle").attr("cx", cx).attr("cy", y(d.err)).attr("r", Math.min(2.2, bw))
      .attr("fill", hit ? col.blue : col.red);
  });
  return svg.node();
}

Today

  • Explain autocorrelation in time and in space
  • Explain why autocorrelated samples produce confidence intervals that are too narrow
  • Tell a sampling design that adds information from one that repeats it
  • Recognize the fix in R: a model that knows the data are correlated

1 Near things are more alike

  • Tobler’s first law: “Everything is related to everything else, but near things are more related than distant things”
  • In time: a warm afternoon predicts a warm evening. One hour later, you learn almost nothing new
  • In space: a sensor predicts its neighbor down the street. A second sensor there adds little
  • This is autocorrelation: a variable correlated with itself at nearby times or places

The usual confidence interval for a mean is

\[\bar{x} \pm t \cdot \frac{s}{\sqrt{n}}\]

  • The \(\sqrt{n}\) assumes each observation brings new, independent information
  • lm(), t.test(), and most AI-written code use this formula by default
  • When observations repeat each other, the formula counts the same information many times

2 In time: a lake temperature logger

  • Goal: the average surface temperature of a lake this summer (June to August)
  • Budget: 20 readings. You choose how far apart they are
  • The lake barely changes from day to night, but it swings with the weather over several days
TipPredict

You take all 20 readings one hour apart. Compared with readings spread over the whole summer, will the 95% CI be wider or narrower? How often will it contain the true summer mean?

T = ({H: 92 * 24, tau: 72, sigma: 1.5, mu: 22, n: 20, tcrit: 2.093})

makeSeries = r => {
  const phi = Math.exp(-1 / T.tau), s = Math.sqrt(1 - phi * phi);
  const x = new Float64Array(T.H);
  let v = rnorm(r) * T.sigma;               // start from the long-run spread
  for (let i = 0; i < T.H; i++) {
    if (i > 0) v = phi * v + s * T.sigma * rnorm(r);
    x[i] = T.mu + v;
  }
  return x;
}

// Hours at which the 20 readings are taken. Evenly spaced readings start at
// a random hour (u in [0, 1)) chosen so the last one still falls in summer.
sampleHours = (u, r, spacing) => {
  if (spacing === "random") {
    const s = new Set();
    while (s.size < T.n) s.add(Math.floor(r() * T.H));
    return [...s].sort((a, b) => a - b);
  }
  const span = (T.n - 1) * spacing;
  const start = Math.floor(u * (T.H - span));
  return d3.range(T.n).map(i => start + i * spacing);
}

// One example summer. The series and the start time use separate seeds, so
// changing the spacing keeps the same summer and roughly the same start.
tExample = {
  const series = makeSeries(rng(100 + tNew));
  const r2 = rng(900 + tNew);
  const hours = sampleHours(r2(), r2, tSpacing);
  const values = hours.map(h => series[h]);
  return {series, hours, values, truth: d3.mean(series), ci: naiveCI(values, T.tcrit)};
}

// 200 more summers with the same design, for the coverage plot
tReps = {
  const r = rng(7 + tNew);
  return d3.range(200).map(() => {
    const series = makeSeries(r);
    const truth = d3.mean(series);
    const values = sampleHours(r(), r, tSpacing).map(h => series[h]);
    const ci = naiveCI(values, T.tcrit);
    return {err: ci.m - truth, half: ci.hi - ci.m, popVar: d3.variance(series)};
  });
}

  • Readings an hour apart all share the same weather. They agree with each other, so \(s\) is small and the CI is tiny
  • But the whole cluster can sit far from the summer mean. The CI misses most of the time
  • Spreading readings out makes the CI wider and honest: wider because it finally sees the summer’s real variability
  • Same \(n\), same formula, very different truth

3 In space: neighborhood heat sensors

  • Goal: the average afternoon air temperature across a 40 × 40 km region
  • Budget: 30 low-cost sensors hosted by volunteers
  • Volunteers tend to live in the same few neighborhoods, so sensors cluster

S = ({L: 40, ell: 5, M: 200, sigma: 1.5, mu: 31, n: 30, tcrit: 2.045})

makeField = r => d3.range(S.M).map(() => [rnorm(r) / S.ell, rnorm(r) / S.ell, r() * 2 * Math.PI])
fieldAt = (f, x, y) => {
  let s = 0;
  for (const [a, b, p] of f) s += Math.cos(a * x + b * y + p);
  return S.mu + S.sigma * Math.sqrt(2 / S.M) * s;
}

// Exact region-wide mean of the surface (the "truth"): each wave's integral
// over the square has a closed form, so no grid approximation is needed.
fieldMean = f => {
  const L = S.L;
  const I = a => Math.abs(a) < 1e-9 ? [L, 0] : [Math.sin(a * L) / a, (1 - Math.cos(a * L)) / a];
  let s = 0;
  for (const [a, b, p] of f) {
    const [ar, ai] = I(a), [br, bi] = I(b);
    const pr = ar * br - ai * bi, pi = ar * bi + ai * br;   // complex product
    s += Math.cos(p) * pr - Math.sin(p) * pi;               // real part of e^{ip} * product
  }
  return S.mu + S.sigma * Math.sqrt(2 / S.M) * s / (L * L);
}

// Keep a cluster of width c fully inside the region
clampCenter = (v, c) => Math.max(c / 2, Math.min(S.L - c / 2, v))

// 30 sensors scattered uniformly inside a c x c square centered on (cx, cy)
placeSensors = (r, cx, cy, c) => d3.range(S.n).map(() => [cx + (r() - 0.5) * c, cy + (r() - 0.5) * c])

// Where the example cluster sits. Clicking the map moves it.
mutable sCenter = [12, 26]

sExample = {
  const f = makeField(rng(300 + sNew));
  const cx = clampCenter(sCenter[0], sSize), cy = clampCenter(sCenter[1], sSize);
  const pts = placeSensors(rng(500), cx, cy, sSize);
  const values = pts.map(([x, y]) => fieldAt(f, x, y));
  return {f, cx, cy, pts, values, truth: fieldMean(f), ci: naiveCI(values, S.tcrit)};
}

// The example surface on an 80 x 80 grid, for drawing the map
sGrid = {
  const g = 80, vals = new Float64Array(g * g);
  for (let j = 0; j < g; j++)
    for (let i = 0; i < g; i++)
      vals[j * g + i] = fieldAt(sExample.f, (i + 0.5) * S.L / g, S.L - (j + 0.5) * S.L / g);
  return {g, vals};
}

// 200 more regions, each with the cluster dropped at a random spot
sReps = {
  const r = rng(11 + sNew);
  return d3.range(200).map(() => {
    const f = makeField(r);
    const truth = fieldMean(f);
    const cx = sSize / 2 + r() * (S.L - sSize), cy = sSize / 2 + r() * (S.L - sSize);
    const values = placeSensors(r, cx, cy, sSize).map(([x, y]) => fieldAt(f, x, y));
    const ci = naiveCI(values, S.tcrit);
    // Spread of one random point in the region, from a coarse 12 x 12 grid
    const grid = [];
    for (let i = 0; i < 12; i++) for (let j = 0; j < 12; j++) grid.push(fieldAt(f, (i + 0.5) * S.L / 12, (j + 0.5) * S.L / 12));
    return {err: ci.m - truth, half: ci.hi - ci.m, popVar: d3.variance(grid)};
  });
}

  • Misses go both ways (too hot and too cold). Clustering doesn’t bias the estimate on average. It makes it far less precise than claimed
  • Once the cluster is wider than the correlation distance (about 10 km here), the sensors start adding real information

4 Why it happens: effective sample size

Two things go wrong at once:

  1. The \(\sqrt{n}\) is too generous. Correlated observations overlap in what they tell you. The effective sample size \(n_\text{eff}\) is smaller than \(n\)
  2. \(s\) is too small. Clustered points see only a slice of the variability (the red dots cover a sliver of the lake’s range)

For readings in a time series where each one correlates \(\rho\) with the next (an AR(1) process), a common approximation is

\[n_\text{eff} \approx n \, \frac{1 - \rho}{1 + \rho}\]

0.99 (one hour apart in the lake) about 1
  • The “worth about” tile in each widget estimates the same idea from the simulations
  • Evenly spaced readings over the whole summer can do better than random, because they never bunch up (try “3 days” vs. “Random times” in the lake widget)

5 In R: a model that knows about the correlation

lm() assumes independent errors. nlme::gls() (generalized least squares) fits the same mean but also estimates how the errors are correlated and widens the CI to match.

# Small helper: one row per model with the estimate, the 95% CI, and the
# CI's width, so the two models are easy to compare at a glance
ci_table <- function(...) {
  models <- list(...)
  rows <- lapply(models, function(m) {
    ci <- confint(m)[1, ]
    c(estimate = unname(coef(m)[1]), lower = unname(ci[1]),
      upper = unname(ci[2]), width = unname(ci[2] - ci[1]))
  })
  round(do.call(rbind, rows), 2)
}

# Simulate the lake many times at several strengths of autocorrelation, and
# fit lm() and gls() to every simulated summer. (This takes ~15 s; freeze
# saves the result, so it only re-runs when this page's source changes.)
set.seed(2027)
rhos   <- c(0, 0.3, 0.6, 0.8, 0.9)   # day-to-day correlation
n_days <- 60                         # readings per summer
n_reps <- 200                        # simulated summers per rho
mu     <- 22                         # true long-run mean (°C)

sim_rows <- list()
for (rho in rhos) {
  for (k in 1:n_reps) {
    # AR(1) series with a total SD of 1.5 °C, whatever rho is
    temp <- if (rho == 0) {
      rnorm(n_days, mu, 1.5)
    } else {
      mu + as.numeric(arima.sim(list(ar = rho), n = n_days,
                                sd = 1.5 * sqrt(1 - rho^2)))
    }
    d <- data.frame(day = 1:n_days, temp = temp)

    # The two models from above
    fits <- list(
      lm  = lm(temp ~ 1, data = d),
      gls = tryCatch(gls(temp ~ 1, data = d, correlation = corAR1(form = ~ day)),
                     error = function(e) NULL)   # skip a rare failed fit
    )
    for (model in names(fits)) {
      if (is.null(fits[[model]])) next
      ci <- confint(fits[[model]])[1, ]
      sim_rows[[length(sim_rows) + 1]] <- data.frame(
        rho = rho, rep = k, model = model,
        est = unname(coef(fits[[model]])[1]), lo = ci[1], hi = ci[2])
    }
  }
}
sim <- do.call(rbind, sim_rows)

# Coverage (share of CIs containing mu) and median CI width, by model and rho.
# Median, not mean: at strong rho a few gls() fits give huge CIs (see below).
sim$covered <- sim$lo <= mu & sim$hi >= mu
sim$width   <- sim$hi - sim$lo
coverage <- merge(
  aggregate(covered ~ model + rho, data = sim, FUN = mean),
  aggregate(width ~ model + rho, data = sim, FUN = median))
coverage$width <- round(coverage$width, 2)
coverage[order(coverage$rho, coverage$model), ]

8 Check yourself

Note1. One air-quality monitor logs PM2.5 every hour for a year (8,760 readings). t.test() gives a mean of 9.1 ± 0.04 µg/m³. Is ± 0.04 believable?
  • No. Hourly readings from one monitor are strongly autocorrelated: pollution episodes last days
  • The effective sample size is far below 8,760, so the true uncertainty about the annual mean is much larger
  • Check with acf(); fit with gls(..., correlation = corAR1()), or average to daily or weekly values first and check those too
Note2. You can afford 50 soil samples to estimate county-wide soil carbon. Design A: 50 samples in one farm. Design B: one sample in each of 50 farms across the county. Which CI can you trust?
  • B. Its samples are far apart, so they’re close to independent and represent the whole county
  • A’s samples describe one farm. Its CI will be narrow and, for the county, very likely wrong (like the 5 km sensor cluster)
Note3. Does autocorrelation make the estimate itself wrong, or only its CI?
  • Mostly the CI. In both widgets, the misses scatter above and below the truth, so the estimate isn’t biased on average
  • But any one clustered estimate can be far off, and the narrow CI hides that
  • If the clusters are placed non-randomly (e.g., volunteers only in cool, leafy neighborhoods), you get bias on top of overconfidence
Note4. An AI fits lm(temp ~ year) to 40 years of monthly temperatures and reports a warming trend with p < 0.001. What do you ask before believing the p-value?
  • Are the residuals autocorrelated? Check acf(resid(model))
  • If yes, refit with gls(temp ~ year, correlation = corAR1()) and compare
  • The trend may still be real, but the p-value from lm() overstates the evidence, for the same reason the CIs above were too narrow
