// Draw one county outline inside a cell. geoIdentity is planar, so the
// ring direction of the polygon doesn't matter (spherical d3 projections
// would fill the whole globe for a counter-clockwise ring).
drawShape = (g, feature, x, y, w, h) => {
  const proj = d3.geoIdentity().reflectY(true)
    .fitExtent([[x + 5, y + 3], [x + w - 5, y + h - 3]], feature);
  g.append("path").attr("d", d3.geoPath(proj)(feature))
    .attr("fill", col.fill).attr("stroke", col.blue).attr("stroke-width", 0.8);
}

// Draw a table into an SVG parent.
//   cols: [{key, label, w}]; rows: array of objects.
//   A "geometry" column draws the county shape (or EMPTY if none);
//   null/undefined values print as a red NA, like R does.
// Each cell group gets the class col-<key>, so a whole column can be faded.
// Rows are positioned with a CSS transform, so they can glide to new slots.
drawTable = (parent, {x, y, cols, rows, rowH = 26, title = null, keyCol = null}) => {
  const g = parent.append("g").attr("transform", `translate(${x},${y})`);
  // Left edge of each column (plus the table's right edge at the end)
  const xs = Array.from(d3.cumsum([0, ...cols.map(c => c.w)]));
  if (title) {
    g.append("text").attr("x", 0).attr("y", -8).attr("font-size", 13)
      .attr("font-weight", 600).attr("fill", col.ink).text(title);
  }
  // Header row (the key column gets an orange header)
  const head = g.append("g").attr("class", "head");
  cols.forEach((c, j) => {
    const cell = head.append("g").attr("class", `col-${c.key}`).style("transition", fade);
    cell.append("rect").attr("x", xs[j]).attr("width", c.w).attr("height", rowH)
      .attr("fill", c.key === keyCol ? col.keyHead : col.head).attr("stroke", col.line);
    cell.append("text").attr("x", xs[j] + 6).attr("y", rowH / 2 + 4.5)
      .attr("font-size", 12.5).attr("font-family", "monospace").attr("font-weight", 600)
      .attr("fill", col.ink).text(c.label ?? c.key);
  });
  // Body rows
  const rowGs = rows.map((r, i) => {
    const rg = g.append("g").attr("class", "row")
      .style("transform", `translate(0px, ${(i + 1) * rowH}px)`).style("transition", glide);
    cols.forEach((c, j) => {
      const cell = rg.append("g").attr("class", `col-${c.key}`).style("transition", fade);
      cell.append("rect").attr("x", xs[j]).attr("width", c.w).attr("height", rowH)
        .attr("fill", "#fff").attr("stroke", col.line);
      const v = r[c.key];
      const txt = cell.append("text").attr("x", xs[j] + 6).attr("y", rowH / 2 + 4.5)
        .attr("font-size", 12.5).attr("font-family", "monospace");
      if (c.key === "geometry") {
        if (v) { txt.remove(); drawShape(cell, v, xs[j], 0, c.w, rowH); }
        else txt.attr("fill", col.muted).attr("font-style", "italic").text("EMPTY");
      } else if (v === null || v === undefined) {
        txt.attr("fill", col.red).attr("font-weight", 700).text("NA");
      } else {
        txt.attr("fill", col.ink).text(v);
      }
    });
    return rg;
  });
  return {g, rowGs, xs, rowH, width: xs[cols.length]};
}

// A curved connector between a left-table row and a right-table row
// ("tentacle"). It starts undrawn: pathLength = 1 makes one dash cover the
// whole curve, so moving stroke-dashoffset from 1 to 0 draws it on.
connector = (parent, x1, y1, x2, y2, color) => parent.append("path")
  .attr("class", "tentacle")
  .attr("d", `M${x1},${y1} C${(x1 + x2) / 2},${y1} ${(x1 + x2) / 2},${y2} ${x2},${y2}`)
  .attr("pathLength", 1)
  .attr("fill", "none").attr("stroke", color).attr("stroke-width", 2)
  .style("stroke-dasharray", "1").style("stroke-dashoffset", "1")
  .style("transition", "stroke-dashoffset 0.9s ease")

An sf object is a data frame with a map attached

  • sf object = a data frame + one geometry column (the shapes)
  • Its class is both: "sf" and "data.frame", so data frame tools still work
  • Sticky geometry: the geometry column stays unless you drop it on purpose (st_drop_geometry())
Data frame sf object
Look Rows and columns Same, plus a header (geometry type, bounding box, CRS) and a geometry column
Feel Every column is independent Geometry sticks to its row through subsets and joins
Functionality dplyr verbs, [ ], $ All of those, plus st_*() functions, maps with plot(), a CRS
Use case Tables with no location: a CDC PLACES CSV, a census table Features on a map: tracts, counties, facilities

2 Subsetting: base R and dplyr

  • Rows by condition: filter() or x[condition, ]
  • Rows by position: slice() or x[1:3, ]
    • Position is fragile: row order can change; a condition says what you mean
  • Columns by name: select() or x[ , c("a", "b")]
  • One column out as a plain vector: x$a or pull(x, a)
    • No geometry: a vector isn’t a table
  • On an sf object, base R and dplyr give the same result

3 Joins: combining tables by a key

Keys

  • Join: add the columns of one table to another by matching rows
  • Key (linking field): the column whose values must match in both tables, e.g. FIPS
  • Rows match by value, not by position: the two tables can be in any order
  • Left table = the first argument; right table = the second

Inner vs. left

  • inner_join(): keep only rows whose key is in both tables
  • left_join(): keep every row of the left table; NA where the right table had no match
  • right_join() and full_join() also keep unmatched rows from the right table, which have no polygon (EMPTY geometry)
  • So with sf: sf object on the left, then left (keep every feature) or inner (keep only matched features)

joinGeom = {
  const leftNames = ["Ashe", "Alleghany", "Surry", "Currituck"];
  const rightNames = ["Surry", "Northampton", "Ashe", "Currituck"];
  // Result rows, in the order dplyr returns them: left rows first, then
  // right-only rows at the end
  const resultRows = [
    ...leftNames.map(n => ({NAME: n, FIPS: county[n].FIPS,
      BIR79: rightNames.includes(n) ? county[n].BIR79 : null, geometry: county[n].shape,
      inLeft: true, inRight: rightNames.includes(n)})),
    {NAME: null, FIPS: county.Northampton.FIPS, BIR79: county.Northampton.BIR79, geometry: null,
     inLeft: false, inRight: true}
  ];
  return {leftNames, rightNames, resultRows};
}

joinSvg = {
  const {leftNames, rightNames, resultRows} = joinGeom;
  const W = 600, H = 410, rowH = 26;
  const svg = d3.create("svg").attr("viewBox", [0, 0, W, H]).style("width", "100%").style("max-width", "700px");
  const L = {x: 10, y: 34}, R = {x: 440, y: 34};
  const left = drawTable(svg, {x: L.x, y: L.y, keyCol: "FIPS", title: "counties (sf)",
    cols: [{key: "NAME", w: 100}, {key: "FIPS", w: 66}, {key: "geometry", w: 70}],
    rows: leftNames.map(n => ({NAME: n, FIPS: county[n].FIPS, geometry: county[n].shape}))});
  const right = drawTable(svg, {x: R.x, y: R.y, keyCol: "FIPS", title: "births79 (data frame)",
    cols: [{key: "FIPS", w: 66}, {key: "BIR79", w: 66}],
    rows: rightNames.map(n => ({FIPS: county[n].FIPS, BIR79: county[n].BIR79}))});
  left.rowGs.forEach((g, i) => g.attr("data-name", leftNames[i]).attr("class", "row lrow"));
  right.rowGs.forEach((g, i) => g.attr("data-name", rightNames[i]).attr("class", "row rrow"));
  // Tentacles between matching keys (drawn on when a join is chosen), and
  // "no match" labels for the rows with nothing to reach
  const lx = L.x + left.width, rx = R.x;
  leftNames.forEach((n, i) => {
    const j = rightNames.indexOf(n);
    const y1 = L.y + (i + 1.5) * rowH;
    if (j >= 0) connector(svg, lx, y1, rx, R.y + (j + 1.5) * rowH, col.green);
    else svg.append("text").attr("class", "nomatch").attr("x", lx + 8).attr("y", y1 + 4).attr("font-size", 12)
      .attr("fill", col.orange).attr("font-weight", 600).style("opacity", 0).style("transition", fade).text("no match");
  });
  rightNames.forEach((n, j) => {
    if (!leftNames.includes(n)) svg.append("text").attr("class", "nomatch").attr("x", rx - 8).attr("y", R.y + (j + 1.5) * rowH + 4)
      .attr("text-anchor", "end").attr("font-size", 12).attr("fill", col.orange).attr("font-weight", 600)
      .style("opacity", 0).style("transition", fade).text("no match");
  });
  // Result table: all five candidate rows; the update cell positions them.

Cardinality: how many matches per key?

  • Cardinality: how many rows on each side share a key value
  • One-to-one: each key once on each side → rows out = rows in
  • One-to-many: a key repeats in the right table → rows multiply, and each polygon repeats
  • dplyr warns only for many-to-many; one-to-many is silent
  • Two right-hand tables for the same three counties:
    • births: births in 1979, one row per county
    • sids_long: SIDS deaths (sudden infant death syndrome) per county per year: one row for 1974 and one for 1979, so two rows per county
  • “Long” format: one row per county and year, instead of one column per year (SID74, SID79 in nc)

Choosing your linking fields

  • Unique on the side that should have one row per key: anyDuplicated(births79$FIPS) returns 0
  • Same meaning: county code ↔︎ county code, not county code ↔︎ some other ID
  • Same type: character vs. numeric is a loud error (below)
  • Same format: "37009" vs. "1400000US37009" is no error, just no matches
  • Codes over names: names vary in spelling and punctuation (“St. Mary’s” vs. “Saint Marys”); codes are built to match
  • Name the key: always write by = "FIPS"; if the names differ, by = c("FIPS" = "fips")
